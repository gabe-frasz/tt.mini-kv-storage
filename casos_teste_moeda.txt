# Casos de Teste - Extensão de Valores Monetários (moeda.lua)
# Formato: Entradas seguidas da saída esperada correspondente

# ---------------------------------------------------------------------------
# 1. Inserções válidas com centavos e formatação correta
# ---------------------------------------------------------------------------
ADD moeda_salario 2500.50
# Esperado: OK
GET moeda_salario
# Esperado: R$ 2.500,50

ADD moeda_preco 12.90
# Esperado: OK
GET moeda_preco
# Esperado: R$ 12,90

# ---------------------------------------------------------------------------
# 2. Inserções válidas com números inteiros (preenchimento de centavos 00)
# ---------------------------------------------------------------------------
ADD moeda_inteiro 100
# Esperado: OK
GET moeda_inteiro
# Esperado: R$ 100,00

ADD moeda_mil 1000
# Esperado: OK
GET moeda_mil
# Esperado: R$ 1.000,00

# ---------------------------------------------------------------------------
# 3. Inserção com 1 dígito decimal (deve completar com zero à direita)
# ---------------------------------------------------------------------------
ADD moeda_dec1 10.5
# Esperado: OK
GET moeda_dec1
# Esperado: R$ 10,50

# ---------------------------------------------------------------------------
# 4. Valores pequenos (centavos puros) e zero
# ---------------------------------------------------------------------------
ADD moeda_centavo 0.05
# Esperado: OK
GET moeda_centavo
# Esperado: R$ 0,05

ADD moeda_um_centavo 0.01
# Esperado: OK
GET moeda_um_centavo
# Esperado: R$ 0,01

ADD moeda_zero 0
# Esperado: OK
GET moeda_zero
# Esperado: R$ 0,00

ADD moeda_zero_dec 0.00
# Esperado: OK
GET moeda_zero_dec
# Esperado: R$ 0,00

# ---------------------------------------------------------------------------
# 5. Valores grandes com múltiplos separadores de milhar
# ---------------------------------------------------------------------------
ADD moeda_milhao 1234567.89
# Esperado: OK
GET moeda_milhao
# Esperado: R$ 1.234.567,89

ADD moeda_bilhao 1000000000.00
# Esperado: OK
GET moeda_bilhao
# Esperado: R$ 1.000.000.000,00

# ---------------------------------------------------------------------------
# 6. Sobrescrita de valor na mesma chave
# ---------------------------------------------------------------------------
ADD moeda_item 50.00
# Esperado: OK
GET moeda_item
# Esperado: R$ 50,00
ADD moeda_item 75.25
# Esperado: OK
GET moeda_item
# Esperado: R$ 75,25

# ---------------------------------------------------------------------------
# 7. Erros: Rejeição de número negativo
# ---------------------------------------------------------------------------
ADD moeda_negativo -10.00
# Esperado: ERRO: valor monetário inválido: deve conter apenas dígitos e ponto decimal opcional

# ---------------------------------------------------------------------------
# 8. Erros: Rejeição de mais de duas casas decimais
# ---------------------------------------------------------------------------
ADD moeda_3casas 10.555
# Esperado: ERRO: valor monetário inválido: máximo de duas casas decimais permitidas

# ---------------------------------------------------------------------------
# 9. Erros: Rejeição de vírgula como separador decimal
# ---------------------------------------------------------------------------
ADD moeda_virgula 10,50
# Esperado: ERRO: formato monetário inválido: utilize ponto como separador decimal

# ---------------------------------------------------------------------------
# 10. Erros: Caracteres alfanuméricos e caracteres especiais
# ---------------------------------------------------------------------------
ADD moeda_letras abc
# Esperado: ERRO: valor monetário inválido: deve conter apenas dígitos e ponto decimal opcional

ADD moeda_simbolo R$10.00
# Esperado: ERRO: valor monetário inválido: deve conter apenas dígitos e ponto decimal opcional

# ---------------------------------------------------------------------------
# 11. Erros: Chave inexistente
# ---------------------------------------------------------------------------
GET moeda_inexistente
# Esperado: ERRO: chave inexistente

EXIT
